use serde_json::{json,Value};
use std::collections::BTreeMap;
use tauri::ipc::{RuntimeAuthority,Origin};
use tauri_utils::{acl::{resolved::Resolved,manifest::Manifest,capability::Capability,Permission,APP_ACL_KEY},platform::Target};
fn run(c:&Value)->Value {
 let mut permissions=BTreeMap::new();
 for (id,allow,deny,scope) in [
  ("allow-narrow",vec!["probe"],vec![],json!({"allow":[{"path":"safe"}],"deny":[{"path":"secret"}]})),
  ("allow-wide",vec!["probe"],vec![],json!({"allow":[{"path":"*"}]})),
  ("deny-probe",vec![],vec!["probe"],json!({})),
  ("deny-other",vec![],vec!["other"],json!({}))] {
  let p:Permission=serde_json::from_value(json!({"identifier":id,"commands":{"allow":allow,"deny":deny},"scope":scope})).unwrap();
  permissions.insert(id.to_string(),p);
 }
 let acl=if c["no_manifest"]==true{BTreeMap::new()}else{[(if c["plugin"]==true{"fixture".to_string()}else{APP_ACL_KEY.to_string()},Manifest{permissions,..Default::default()})].into_iter().collect()};
 let caps:BTreeMap<String,Capability>=c["caps"].as_array().unwrap().iter().map(|v|{let cap:Capability=serde_json::from_value(v.clone()).unwrap();(cap.identifier.clone(),cap)}).collect();
 let target:Target=serde_json::from_value(c["target"].clone()).unwrap();
 let mut r=match Resolved::resolve(&acl,if c["dynamic"]==true{BTreeMap::new()}else{caps.clone()},target){Ok(r)=>r,Err(e)=>return json!({"stage":"resolve-error","error":e.to_string()})};
 if c["empty_deny"]==true{r.denied_commands.insert("probe".into(),vec![]);}
 let scopes=format!("{:?}",r.command_scope);
 let origin=if c["local"]==true{Origin::Local}else{Origin::Remote{url:url::Url::parse(c["url"].as_str().unwrap()).unwrap()}};
 let canonical=match &origin {Origin::Local=>"local".to_string(),Origin::Remote{url}=>url.to_string()};
 let has_app_manifest=r.has_app_acl; let mut a=RuntimeAuthority::new(acl,r); if c["dynamic"]==true { for cap in caps.values() { if let Err(e)=a.add_capability(serde_json::to_string(cap).unwrap()){return json!({"stage":"add-error","error":e.to_string()});} } }
 let access=a.resolve_access(c["command"].as_str().unwrap(),c["window"].as_str().unwrap(),c["webview"].as_str().unwrap(),&origin);
 let entries=access.as_ref().map(|cmds|cmds.iter().map(|cmd|json!({"scope_id":cmd.scope_id,"capability":cmd.referenced_by.capability,"permission":cmd.referenced_by.permission,"context":format!("{:?}",cmd.context)})).collect::<Vec<_>>()).unwrap_or_default();
 json!({"stage":"runtime","allowed":access.is_some(),"entries":entries,"scope_store":scopes,"has_app_manifest":has_app_manifest,"canonical_url":canonical})
}
fn main(){std::panic::set_hook(Box::new(|_|{}));let cases:Vec<Value>=serde_json::from_str(include_str!("../../cases.json")).unwrap();
 let out:Vec<_>=cases.iter().map(|c|{let r=std::panic::catch_unwind(std::panic::AssertUnwindSafe(||run(c))).unwrap_or_else(|e|{let msg=e.downcast_ref::<String>().map(String::as_str).or(e.downcast_ref::<&str>().copied()).unwrap_or("panic");json!({"stage":"panic","error":msg})});json!({"id":c["id"],"result":r})}).collect();println!("{}",json!(out));}
