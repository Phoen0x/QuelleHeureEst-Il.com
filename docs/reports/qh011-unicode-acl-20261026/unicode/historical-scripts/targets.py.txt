import subprocess,json,concurrent.futures
from pathlib import Path
w=Path('<UNICODE_DIAGNOSTIC>')
def run(t):
 side,target,all_features=t
 cmd=['cargo','metadata','--locked','--offline','--format-version','1','--filter-platform',target]
 if all_features:cmd+=['--all-features']
 p=subprocess.run(cmd,cwd=w/side,capture_output=True,text=True,timeout=60)
 row={'side':side,'target':target,'all_features':all_features,'exit':p.returncode}
 if p.returncode:row['error']=p.stderr;return row
 d=json.loads(p.stdout);(w/('filtered-'+side+'-'+target+('-all' if all_features else '')+'.json')).write_text(p.stdout)
 packages={x['id']:x for x in d['packages']};active={packages[n['id']]['name']+'@'+packages[n['id']]['version'] for n in d['resolve']['nodes']}
 row.update(count=len(active),unic=sorted(x for x in active if x.startswith('unic-')),glib=sorted(x for x in active if x.startswith('glib@')),pem='rustls-pemfile@1.0.4' in active)
 return row
tasks=[(s,t,a) for s in ['before','after'] for t in ['x86_64-unknown-linux-gnu','aarch64-linux-android','x86_64-pc-windows-msvc','aarch64-apple-darwin','aarch64-apple-ios','wasm32-unknown-unknown'] for a in [False,True]]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as ex:rows=list(ex.map(run,tasks))
(w/'target-summary.json').write_text(json.dumps(rows,indent=2));print(json.dumps(rows))
