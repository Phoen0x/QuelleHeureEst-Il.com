//! FR : sonde expérimentale de construction, pas une couche ACL intégrée.
//! EN: Experimental construction probe, not an integrated ACL layer.
use serde_json::{json, Value};
use std::collections::BTreeMap;
use tauri::ipc::{Origin, RuntimeAuthority};
use tauri_utils::{acl::{capability::Capability, manifest::Manifest, resolved::Resolved, RemoteUrlPattern}, platform::Target};

#[derive(Debug, PartialEq)]
pub enum BuildError { UnknownVersion, HistoricalRemoteUnproven, ReviewRequired, ReviewMismatch, InvalidConfiguration(String), InvalidPattern(String), Resolution(String), ResolutionPanicked }

// FR : l'instantané lie toutes les permissions/scopes, pas seulement les URL.
// EN: The snapshot binds all permissions/scopes, not just URLs.
pub fn build_policy(envelope: &Value, reviewed_snapshot: Option<&Value>) -> Result<(BTreeMap<String, Manifest>, Resolved), BuildError> {
    let config=&envelope["configuration"];
    let manifests:BTreeMap<String,Manifest>=serde_json::from_value(config["manifests"].clone()).map_err(|e|BuildError::InvalidConfiguration(e.to_string()))?;
    let caps:BTreeMap<String,Capability>=serde_json::from_value(config["capabilities"].clone()).map_err(|e|BuildError::InvalidConfiguration(e.to_string()))?;
    let target:Target=serde_json::from_value(config["target"].clone()).map_err(|e|BuildError::InvalidConfiguration(e.to_string()))?;
    match envelope["version"].as_str() {
        Some("legacy-urlpattern-0.3") => {
            // FR : sans preuve de langage, refuser la construction entière.
            // EN: Without a language proof, reject the entire construction.
            if caps.values().any(|c|c.remote.as_ref().is_some_and(|r|!r.urls.is_empty())) { return Err(BuildError::HistoricalRemoteUnproven); }
        }
        Some("reviewed-urlpattern-0.6") => {
            let review=reviewed_snapshot.ok_or(BuildError::ReviewRequired)?;
            if review!=envelope {return Err(BuildError::ReviewMismatch);}
        }
        _=>return Err(BuildError::UnknownVersion)
    }
    for cap in caps.values() {
        if let Some(remote)=&cap.remote {for pattern in &remote.urls {
            pattern.parse::<RemoteUrlPattern>().map_err(|e|BuildError::InvalidPattern(e.to_string()))?;
        }}
    }
    let resolved=std::panic::catch_unwind(std::panic::AssertUnwindSafe(||Resolved::resolve(&manifests,caps,target)))
        .map_err(|_|BuildError::ResolutionPanicked)?.map_err(|e|BuildError::Resolution(e.to_string()))?;
    Ok((manifests,resolved))
}

mod tests {

fn cases()->Vec<Value>{serde_json::from_str(include_str!("../fixtures/cases.json")).unwrap()}
fn case(id:&str)->Value{cases().into_iter().find(|x|x["id"]==id).unwrap()}
fn envelope(c:&Value,version:&str)->Value {
    let mut permissions=serde_json::Map::new();
    for (id,allow,deny,scope) in [
        ("allow-narrow",vec!["probe"],vec![],json!({"allow":[{"path":"safe"}],"deny":[{"path":"secret"}]})),
        ("allow-wide",vec!["probe"],vec![],json!({"allow":[{"path":"*"}]})),
        ("deny-probe",vec![],vec!["probe"],json!({})),
        ("deny-other",vec![],vec!["other"],json!({}))] {
        permissions.insert(id.into(),json!({"identifier":id,"commands":{"allow":allow,"deny":deny},"scope":scope}));
    }
    let mut manifests=serde_json::Map::new();
    if c["no_manifest"]!=true {manifests.insert(if c["plugin"]==true{"fixture"}else{APP_ACL_KEY}.into(),json!({"permissions":permissions,"permission_sets":{},"default_permission":null,"global_scope_schema":null}));}
    let mut caps=serde_json::Map::new();for cap in c["caps"].as_array().unwrap(){caps.insert(cap["identifier"].as_str().unwrap().into(),cap.clone());}
    json!({"version":version,"configuration":{"manifests":manifests,"capabilities":caps,"target":c["target"]}})
}
fn modern(id:&str)->Value{envelope(&case(id),"reviewed-urlpattern-0.6")}
fn observation(c:&Value,mut r:Resolved,manifests:BTreeMap<String,Manifest>)->Value{
    if c["empty_deny"]==true {r.denied_commands.insert("probe".into(),vec![]);}
    let scopes=format!("{:?}",r.command_scope);let global=format!("{:?}",r.global_scope);
    let authority=RuntimeAuthority::new(manifests,r);
    let origin=if c["local"]==true{Origin::Local}else{Origin::Remote{url:url::Url::parse(c["url"].as_str().unwrap()).unwrap()}};
    let result=authority.resolve_access(c["command"].as_str().unwrap(),c["window"].as_str().unwrap(),c["webview"].as_str().unwrap(),&origin);
    let entries=result.as_ref().map(|list|list.iter().map(|v|json!({"scope":v.scope_id,"capability":v.referenced_by.capability,"permission":v.referenced_by.permission})).collect::<Vec<_>>()).unwrap_or_default();
    json!({"allowed":result.is_some(),"entries":entries,"scopes":scopes,"global":global})
}
